# Comandos de Instalação para Publicidade Player no Raspberry Pi OS

# 1. Atualizar a lista de pacotes do sistema
sudo apt update

# 2. Instalar Python 3 e Pip (geralmente já vêm, mas garante)
sudo apt install python3 python3-pip -y

# 3. Instalar PyQt5 e o módulo QtMultimedia via apt (essencial)
sudo apt install python3-pyqt5 python3-pyqt5.qtmultimedia -y

# 4. Instalar/Reinstalar os plugins GStreamer (crucial para vídeo)
# Usar --reinstall pode ajudar a corrigir problemas de instalação anterior
sudo apt install --reinstall gstreamer1.0-plugins-base gstreamer1.0-plugins-good gstreamer1.0-plugins-bad gstreamer1.0-plugins-ugly gstreamer1.0-libav libqt5multimedia5-plugins -y

# 5. Instalar a biblioteca Python 'requests' via pip (para a API)
pip3 install requests

# Após executar todos esses comandos, navegue até o diretório
# onde está o script 'main.py' e execute com:
# python3 main.py
